```sql
-- 1
SELECT *
FROM Employees
WHERE salary > (SELECT AVG(salary) FROM Employees);

-- 2
SELECT *
FROM Employees
WHERE salary = (SELECT MAX(salary) FROM Employees);

-- 3
SELECT *
FROM Employees
WHERE salary > (
    SELECT MAX(salary)
    FROM Employees
    WHERE emp_name = 'Ravi'
);

-- 4
SELECT *
FROM Employees
WHERE department IN (
    SELECT department
    FROM Employees
    WHERE emp_name = 'Kiran'
);

-- 5
SELECT MAX(salary) AS second_highest_salary
FROM Employees
WHERE salary < (SELECT MAX(salary) FROM Employees);

-- 6
SELECT *
FROM Employees e
WHERE salary = (
    SELECT MIN(salary)
    FROM Employees
    WHERE department = e.department
);

-- 7
SELECT DISTINCT salary
FROM Employees
ORDER BY salary DESC
LIMIT 1 OFFSET 2;

-- 8
SELECT *
FROM Employees
ORDER BY salary DESC
LIMIT 3;

-- 9
SELECT salary, COUNT(*) AS employee_count
FROM Employees
GROUP BY salary
HAVING COUNT(*) > 1;

-- 10
UPDATE Employees
SET salary = salary * 1.10
WHERE department = 'IT';

-- 11
DELETE FROM Employees
WHERE salary < 30000;

-- 12
SELECT UPPER(emp_name) AS employee_name
FROM Employees;

-- 13
SELECT *
FROM Employees
WHERE LOWER(emp_name) LIKE '%a%';

-- 14
SELECT
    emp_id,
    emp_name,
    salary,
    department,
    CASE
        WHEN salary >= 50000 THEN 'High'
        WHEN salary >= 30000 THEN 'Medium'
        ELSE 'Low'
    END AS salary_grade
FROM Employees;
```
